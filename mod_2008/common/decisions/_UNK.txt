player_assistance = {

	player_assist_heavy = {
		available = {
			is_ai = no
		}
		complete_effect = {
			remove_ideas = { player_assist_light_idea player_assist_medium_idea } add_ideas = player_assist_heavy_idea
		}
	}

	player_assist_light = {
		available = {
			is_ai = no
		}
		complete_effect = {
			remove_ideas = { player_assist_medium_idea player_assist_heavy_idea } add_ideas = player_assist_light_idea
		}
	}

	player_assist_medium = {
		available = {
			is_ai = no
		}
		complete_effect = {
			remove_ideas = { player_assist_light_idea player_assist_heavy_idea } add_ideas = player_assist_medium_idea
		}
	}

	player_assist_reset = {
		available = {
			is_ai = no
		}
		visible = {
			OR = { has_idea = player_assist_light_idea } OR = { has_idea = player_assist_medium_idea } OR = { has_idea = player_assist_heavy_idea }
		}
		complete_effect = {
			remove_ideas = { player_assist_light_idea player_assist_medium_idea player_assist_heavy_idea }
		}
	}

}
